async def process_query(client: Client, query: str) -> str:
    """Process a query using Claude and available tools."""
    messages = [{"role": "user", "content": query}]

    # Advertise every tool exposed by the MCP server
    tool_list = await client.list_tools()
    available_tools = [{
        "name": tool.name,
        "description": tool.description,
        "input_schema": tool.inputSchema
    } for tool in tool_list.tools]

    response = anthropic.messages.create(
        model=MODEL, max_tokens=1000,
        messages=messages, tools=available_tools
    )

    final_text, tool_results = [], []
    for content in response.content:
        if content.type == "text":
            final_text.append(content.text)
        elif content.type == "tool_use":
            result = await client.call_tool(content.name, content.input)
            result_text = "\n".join(block.text for block in result.content
                                    if isinstance(block, TextContent))
            if result.isError:
                result_text = f"Tool execution failed: {result_text or 'Unknown error'}"
            tool_results.append({
                "type": "tool_result",
                "tool_use_id": content.id,
                "content": result_text
            })

    if tool_results:
        messages.extend([
            {"role": "assistant", "content": response.content},
            {"role": "user", "content": tool_results}
        ])
        response = anthropic.messages.create(
            model=MODEL, max_tokens=1000,
            messages=messages, tools=available_tools
        )
        final_text.extend(block.text for block in response.content
                          if block.type == "text")

    return "\n".join(final_text)
01 — DISCOVER

List capabilities

Fetch tool names, descriptions, and input schemas directly from the connected MCP server.

02 — EXECUTE

Run requested tools

Match each tool-use block to the client and preserve its unique ID for the response.

03 — RESOLVE

Return grounded context

Send structured tool results back so Claude can compose a useful, evidence-based answer.
